# 2026-10-29（P0）：新增 Disallow /api/。
# 起因：14 天日志实测，爬虫 24,846 次请求里有 12,106 次（48.7%）打在 /api/ 上
#   /api/v1/analytics/events 2838 次、/api/v1/auth/guest-login 1463 次 …
# 现代爬虫会渲染 JS 并跟着执行页面里的 fetch；不声明就会持续消耗爬虫预算。
# 对新站尤其要紧——配额有限，浪费在接口上等于少抓内容页。
# 一并禁 /admin/（管理后台不该进索引）。

User-agent: *
Allow: /
Disallow: /api/
Disallow: /admin/
# 2026-11-12（P0）：挡 JS 分包，只放行 CSS。
# 依据：10-31~11-12 实测真实 Baiduspider 49 次请求中 29 次（59%）打在 /_next/static/，
#   内容页 0 次。页面正文 100% 在 SSR HTML 里（个股页可见文本 517 字，已实测），
#   爬虫拿不到 JS 不影响收录。
# 全站唯一 CSS 是 /_next/static/chunks/<hash>.css，hash 每次构建都变，
#   所以必须按 *.css 通配放行，不能写死文件名。全挡会让爬虫拿到无样式页。
Disallow: /_next/static/
Allow: /_next/static/chunks/*.css
